% Written to the .toc immediately before the first appendix entry. This keeps
% appendix labels correct while the contents file is read on the next build.
\newcommand{\ntust@tocappendixmatter}{\ntust@tocappendixtrue}

\newcommand{\ntust@chapterlabel}{%
  \ifntust@appendix
    \appendixname\space\thechapter
  \else
    \ifdefstring{\ntust@language}{zh}
      {第\zhnumber{\arabic{chapter}}章}
      {\chaptername\space\thechapter}%
  \fi
}

  \ifdefstring{\ntust@language}{zh}{%
    \renewcommand{\contentsname}{目錄}%
    \renewcommand{\listfigurename}{圖目錄}%
    \renewcommand{\listtablename}{表目錄}%
    \renewcommand{\figurename}{圖}%
    \renewcommand{\tablename}{表}%
    \renewcommand{\chaptername}{第}%
    \renewcommand{\bibname}{參考文獻}%
    \renewcommand{\appendixname}{附錄}%
    \renewcommand{\thechapter}{\arabic{chapter}}%
    \renewcommand{\thesection}{\zhnumber{\arabic{section}}、}%
    \renewcommand{\thesubsection}{（\zhnumber{\arabic{subsection}}）}%
    \renewcommand{\thesubsubsection}{\arabic{subsubsection}.}%
    \renewcommand{\theparagraph}{（\arabic{paragraph}）}%
    \setlength{\parindent}{2em}%
  }{%
    \renewcommand{\contentsname}{Contents}%
    \renewcommand{\listfigurename}{List of Figures}%
    \renewcommand{\listtablename}{List of Tables}%
    \renewcommand{\figurename}{Figure}%
    \renewcommand{\tablename}{Table}%
    \renewcommand{\chaptername}{Chapter}%
    \renewcommand{\bibname}{References}%
    \renewcommand{\appendixname}{Appendix}%
    \renewcommand{\thechapter}{\arabic{chapter}}%
    \renewcommand{\thesection}{\thechapter.\arabic{section}}%
    \renewcommand{\thesubsection}{\thesection.\arabic{subsection}}%

  \titleformat{\chapter}[block]

  \renewcommand{\thealgorithm}{\thechapter-\arabic{algorithm}}%
}

\renewcommand*\l@chapter[2]{%
  \begingroup
  \ifdefstring{\ntust@language}{en}{%
    \ifntust@tocappendix
      \def\ntust@tocentryprefix{\appendixname\space}%
      \def\ntust@tocentrynumwidth{5.0em}%
    \else
      \def\ntust@tocentryprefix{\chaptername\space}%
      \def\ntust@tocentrynumwidth{4.8em}%
    \fi
    \renewcommand{\numberline}[1]{\hb@xt@\@tempdima{\ntust@tocentryprefix##1\space\hfil}}%
    \@dottedtocline{0}{0em}{\ntust@tocentrynumwidth}{#1}{#2}%
  }{%
    \@dottedtocline{0}{0em}{2.5em}{#1}{#2}%
  }%
  \endgroup
}
